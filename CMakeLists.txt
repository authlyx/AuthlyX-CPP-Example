cmake_minimum_required(VERSION 3.24)
project(AuthlyX VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(AuthlyX src/AuthlyX.cpp)
target_include_directories(AuthlyX PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(AuthlyX PUBLIC AUTHLYX_SEPARATE_IMPLEMENTATION)
target_link_libraries(AuthlyX PUBLIC OpenSSL::Crypto CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
  target_link_libraries(AuthlyX PUBLIC crypt32 advapi32)
endif()
if(MSVC)
  target_compile_options(AuthlyX PUBLIC /utf-8 /W4)
endif()
add_executable(AuthlyX-CPP-Example AuthlyX-CPP-Example/main.cpp)
target_link_libraries(AuthlyX-CPP-Example PRIVATE AuthlyX)
target_precompile_headers(AuthlyX-CPP-Example PRIVATE AuthlyX.h)
option(AUTHLYX_BUILD_TESTS "Build protocol tests" OFF)
if(AUTHLYX_BUILD_TESTS)
  add_executable(header-check ../tests/cpp_header_a.cpp ../tests/cpp_header_b.cpp)
  target_link_libraries(header-check PRIVATE AuthlyX)
  add_executable(header-only-check ../tests/cpp_header_a.cpp ../tests/cpp_header_b.cpp)
  target_include_directories(header-only-check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(header-only-check PRIVATE OpenSSL::Crypto CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_link_libraries(header-only-check PRIVATE crypt32 advapi32)
  endif()
  add_executable(conformance ../tests/cpp_smoke.cpp)
  target_link_libraries(conformance PRIVATE AuthlyX)
endif()
